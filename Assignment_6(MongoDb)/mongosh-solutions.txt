// --- Task 1: Create explicit collection 'books' with validation rule ---

use ASSIGNMENT_7_DEV

db.createCollection("books", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["title"],
      properties: {
        title: {
          bsonType: "string",
          minLength: 1,
          description: "title must be a non-empty string"
        }
      }
    }
  }
})

// --- Task 15: Delete all books published before 2000 ---

use ASSIGNMENT_7_DEV

db.books.deleteMany({ year: { $lt: 2000 } })



// --- Task 16: Using aggregation functions, filter books published after 2000 and sort by year descending ---

use ASSIGNMENT_7_DEV

db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $sort: { year: -1 } }
])


// --- Task 17: Find books published after 2000 showing only title, author, and year ---

use ASSIGNMENT_7_DEV

db.books.aggregate([
  { $match: { year: { $gt: 2000 } } },
  { $project: { _id: 0, title: 1, author: 1, year: 1 } }
])



// --- Task 18: Break an array of genres into separate documents ---

use ASSIGNMENT_7_DEV

db.books.aggregate([
  { $unwind: "$genres" },
  { $project: { _id: 0, title: 1, genres: 1 } }
])



// --- Task 19: Join logs collection with books collection ---

use ASSIGNMENT_7_DEV

db.logs.aggregate([
  {
    $addFields: {
      bookObjectId: { $toObjectId: "$book_id" }
    }
  },
  {
    $lookup: {
      from: "books",
      localField: "bookObjectId",
      foreignField: "_id",
      as: "book_details"
    }
  },
  {
    $project: {
      _id: 0,
      action: 1,
      "book_details.title": 1,
      "book_details.author": 1,
      "book_details.year": 1
    }
  }
])
